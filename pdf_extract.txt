import jwt, requests, json
from django.conf import settings
from django.contrib.auth.models import User
def _get_jwks():
r = requests.get(settings.CLERK_JWKS_URL, timeout=10)
return r.json()
def verify_clerk_token(token: str) -> dict | None:
try:
jwks = _get_jwks()
return jwt.decode(token, jwks, algorithms=['RS256'],
options={'verify_aud': False})
except Exception:
return None
def get_or_create_user(payload: dict) -> User:
clerk_id = payload['sub']
email = payload.get('email', '')
first_name = payload.get('first_name', '')
last_name = payload.get('last_name', '')
try:
from apps.accounts.models import UserProfile
return UserProfile.objects.get(clerk_user_id=clerk_id).user
except UserProfile.DoesNotExist:
pass
user, _ = User.objects.get_or_create(
email=email, defaults={'username': email,
'first_name': first_name,
'last_name': last_name})
user.profile.clerk_user_id = clerk_id
user.profile.email_verified = True
user.profile.save()
return user
class ClerkAuthMiddleware:
def __init__(self, get_response): self.get_response = get_response
def __call__(self, request):
auth = request.headers.get('Authorization', '')
if auth.startswith('Bearer '):
payload = verify_clerk_token(auth[7:])
if payload: request.user = get_or_create_user(payload)
return self.get_response(request)

def send_messages(self, email_messages):
sent = 0
for msg in email_messages:
try:
body = getattr(msg, 'alternatives', None)
html = body[0][0] if body else f'{msg.body}'

O Clerk cuida do registro, login e OAuth. O Django cuida dos dados de negócio (perfil, plano, equipe, permissões). O

■ Fluxo de autenticação
# Fluxo de autenticação — passo a passo:
#
# 1. Usuário acessa /login → Clerk exibe formulário (email/senha ou Google/GitHub)
# 2. Clerk autentica e retorna JWT assinado para o browser
# 3. Frontend envia JWT em TODAS as requisições:
# Header: Authorization: Bearer eyJhbGci...
# 4. ClerkAuthMiddleware do Django:
# a. Extrai o token do header Authorization
# b. Busca chaves públicas do Clerk (JWKS endpoint)
# c. Verifica a assinatura do JWT com as chaves
# d. Extrai user_id (campo 'sub') do payload
# e. Busca ou cria o User Django com aquele clerk_user_id
# f. Define request.user → todas as views funcionam normalmente
#
# 5. Views Django funcionam como sempre:
# @login_required → funciona (verifica request.user.is_authenticated)
# request.user.profile → dados do perfil
# request.user.social_accounts.all() → contas conectadas
# request.user.profile.is_plan_active → plano válido?

# apps/accounts/permissions.py
from functools import wraps
from django.shortcuts import redirect
from django.contrib import messages
from .models import TeamMembership
ROLE_PERMISSIONS = {
'owner': ['post.*','account.*','team.*','analytics.view','billing.*'],
'admin': ['post.*','account.connect','team.invite','analytics.view'],
'editor': ['post.create','post.edit','post.publish','analytics.view'],
'viewer': ['analytics.view'],
}
def has_permission(user, team, perm: str) -> bool:
try:
m = TeamMembership.objects.get(user=user, team=team, invite_accepted=True)
allowed = ROLE_PERMISSIONS.get(m.role, [])
# Suporte a wildcard: 'post.*' cobre 'post.create', 'post.edit', etc.
for a in allowed:
if a == perm: return True
if a.endswith('.*') and perm.startswith(a[:-1]): return True
return False
except TeamMembership.DoesNotExist:
return False

return view_func(request, *args, **kwargs)

# apps/scheduler/tasks.py
from celery import shared_task
from django.utils import timezone
from .models import ScheduledPost, PublicationResult
from apps.social.services.twitter import TwitterService
@shared_task(bind=True, max_retries=3)
def publish_scheduled_posts(self):
"""Roda a cada 60s — publica posts cujo horário chegou"""
now = timezone.now()
posts = ScheduledPost.objects.filter(
status='scheduled',
scheduled_at__lte=now,
).prefetch_related('accounts')
for post in posts:
post.status = 'publishing'
post.save(update_fields=['status'])
for account in post.accounts.filter(is_active=True):
result = PublicationResult(post=post, account=account)
try:
# Seleciona o serviço correto pela plataforma
service = get_service(account)
text = post.full_text_for(account.platform)
res = service.post_text(text)
if res['success']:
result.status = 'success'
result.platform_post_id = res['post_id']
result.platform_post_url = res.get('url', '')
else:
result.status = 'failed'
result.error_message = res['error']
except Exception as exc:
result.status = 'failed'
result.error_message = str(exc)
# Celery vai tentar novamente (max 3x)
raise self.retry(exc=exc, countdown=300) # 5 min
result.save()
# Marcar o post como publicado ou falhou
all_results = post.results.all()
if all_results.filter(status='success').exists():
post.status = 'published'
post.published_at = timezone.now()
else:
post.status = 'failed'
post.save(update_fields=['status','published_at'])

from apps.ai_engine.services import generate_reply
def process_message(user: User, platform: str, sender: str,
message_text: str, message_obj: InboxMessage) -> str | None:
"""
Processa uma mensagem recebida e envia resposta automática se aplicável.
Retorna o texto da resposta, ou None se não respondeu.
"""
from django.utils import timezone
rules = AutoReplyRule.objects.filter(
user=user, is_active=True,
applies_to__in=[platform, 'both']
)
for rule in rules:
# Verificar se está dentro do horário ativo
hour = timezone.now().hour
if not (rule.active_hours_start <= hour <= rule.active_hours_end):
continue
# Verificar limite de respostas
today = timezone.now().date()
replies_today = InboxReply.objects.filter(
message__user=user,
is_auto_reply=True,
created_at__date=today
).count()
if replies_today >= rule.max_per_day:
continue
# Verificar trigger
triggered = False
if rule.trigger_type == 'all':
triggered = True
elif rule.trigger_type == 'keyword':
keywords = [k.strip().lower() for k in rule.trigger_keywords.split(',')]
triggered = any(kw in message_text.lower() for kw in keywords)
if not triggered:
continue
# Gerar resposta
if rule.reply_mode == 'fixed':
reply_text = rule.fixed_reply
else:
reply_text = generate_reply(
user_message=message_text,
context=rule.ai_instructions,
tone=rule.ai_tone,
)
# Salvar resposta no banco
InboxReply.objects.create(
message=message_obj,
content=reply_text,
is_auto_reply=True,
status='sent',
)
message_obj.is_replied = True
message_obj.auto_replied = True
message_obj.save(update_fields=['is_replied','auto_replied'])
rule.reply_count += 1
